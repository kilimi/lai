# =============================================================================
#
#   python -m venv .venv
#   .venv\Scripts\activate          # Windows
#   source .venv/bin/activate       # Linux / macOS

#
#   conda activate lai_local

#   pip install -r tests/python/requirements.txt

# =============================================================================
#
#   pytest tests/python/
#   pytest tests/python -q -m "not training_smoke and not insid3_smoke"
#
# INSID3 mask helpers (host, no GPU):
#   pytest tests/python/test_insid3_mask_utils.py tests/python/test_insid3_same_image.py::test_same_image_reference_mask_aligns_with_image_b64 -v
